SELECT
  sex,
  passenger_class,
  COUNT(*) AS total_passengers,
  ROUND(100.0 * AVG(survived), 2) AS survival_rate_pct
FROM passengers
GROUP BY sex, passenger_class
ORDER BY sex, passenger_class;
